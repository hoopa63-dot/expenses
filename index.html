function editOption(selectId, labelName) { const targetSelect = document.getElementById(selectId); const selectedOption = targetSelect.options[targetSelect.selectedIndex]; if (!selectedOption || selectedOption.disabled) { alert(`يرجى اختيار ${labelName} لتعديله أولاً.`); return; } const oldValue = selectedOption.value; const newValue = prompt(`تعديل اسم ${labelName}:`, oldValue); if (newValue && newValue.trim() !== '' && newValue.trim() !== oldValue) { const updatedValue = newValue.trim(); // 1. تحديث القائمة المنسدلة في نافذة الإدخال selectedOption.value = updatedValue; selectedOption.textContent = updatedValue; // 2. تحديث القائمة المنسدلة في نافذة التقارير let reportSelectId = ''; if (selectId === 'categorySelect') reportSelectId = 'reportCategory'; if (selectId === 'citySelect') reportSelectId = 'reportCity'; if (selectId === 'deptSelect') reportSelectId = 'reportDept'; if (selectId === 'paySelect') reportSelectId = 'reportPayMethod'; if (reportSelectId) { const reportSelect = document.getElementById(reportSelectId); for (let opt of reportSelect.options) { if (opt.value === oldValue) { opt.value = updatedValue; opt.textContent = updatedValue; break; } } } // 3. تحديث البيانات المسجلة سابقاً في الجدول expensesData.forEach(item => { if (selectId === 'categorySelect' && item.category === oldValue) item.category = updatedValue; if (selectId === 'citySelect' && item.city === oldValue) item.city = updatedValue; if (selectId === 'deptSelect' && item.dept === oldValue) item.dept = updatedValue; if (selectId === 'paySelect' && item.payMethod === oldValue) item.payMethod = updatedValue; }); renderTable(currentlyDisplayedData); } }
